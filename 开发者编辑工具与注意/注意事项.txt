nginx.conf (或你的主配置文件中 http 块的片段)：包含 map 指令。
wallpaper.kivotos.de.conf (或你的网站 server 块文件)：包含 server 块的详细配置。
1. nginx.conf (或主配置文件中 http 块的片段)
你需要在主 NGINX 配置文件 (/etc/nginx/nginx.conf) 或其通过 include 引入的任何文件中找到 http { ... } 块，并在其内部添加以下 map 配置。（这里是在其内部添加map配置）

具体配置代码：
# 通常在 /etc/nginx/nginx.conf 文件中找到 'http' 块
http {
    # ... 其他 http 块的配置，例如 mime.types, sendfile, keepalive_timeout 等 ...

    # ================================================
    # Map 指令用于定义重定向规则 - 放置在 http 块内
    # ================================================
    map $request_uri $redirect_students_url {
        default ""; # 默认不重定向

        # 将旧的编码路径 "/students/佳代子" 重定向到新的路径
        # 注意：这里直接使用 URL 编码后的路径
        "/students/%E4%BD%B3%E4%BB%A3%E5%AD%90" "/students/kayoko/original/index.html"; # ----------》主要修改动这里要将students.html出来的url和live2d的url对接 例子下面给了
        # 未来如果还有更多一对一的精确重定向，可以在此处继续添加：
        # "/old-path-A" "/new-path-X";
        # "/old-path-B" "/new-path-Y";
    }
    # ================================================

    # ... 其他 http 块的配置 ...

    # 包含你的 server 块文件，例如：
    include /etc/nginx/conf.d/*.conf;
}

分析一下 最重要的一句 "/students/%E4%BD%B3%E4%BB%A3%E5%AD%90" "/students/kayoko/original/index.html";

2. wallpaper.kivotos.de.conf (你的网站 server 块文件)
这个文件应该包含了你网站 server 块的所有详细配置。我已经将重定向逻辑添加进去了。

具体代码：
# 这是一个单独的 server 配置文件，例如位于 /etc/nginx/conf.d/ 或 /www/server/panel/vhost/nginx/
server
{
    listen 80;
    listen 443 ssl;
    listen 443 quic;
    listen [::]:443 ssl;
    listen [::]:443 quic;
    http2 on;
    listen [::]:80;
    server_name wallpaper.kivotos.de;
    index students.html index.php index.htm default.php default.htm default.html;
    root /www/wwwroot/wallpaper.kivotos.de;

    # CERT-APPLY-CHECK--START
    # 用于SSL证书申请时的文件验证相关配置 -- 请勿删除
    include /www/server/panel/vhost/nginx/well-known/wallpaper.kivotos.de.conf;
    # CERT-APPLY-CHECK--END

    # SSL-START SSL相关配置，请勿删除或修改下一行带注释的404规则
    #error_page 404/404.html;
    #HTTP_TO_HTTPS_START##############################################----------》其实重点从这就开始了
    set $isRedcert 1;
    if ($server_port != 443) {
        set $isRedcert 2;
    }
    if ( $uri ~ /\.well-known/ ) {
        set $isRedcert 1;
    }
    if ($isRedcert != 1) {
        # 这个 rewrite 会优先处理 HTTP 到 HTTPS 的重定向。
        rewrite ^(/.*)$ https://$host$1 permanent;
    }
    #HTTP_TO_HTTPS_END

    # ================================================			#
    # 新增的重定向规则 - 放置在 HTTP_TO_HTTPS_END 之后			#
    # ================================================			#
    # 检查 $redirect_students_url 变量，如果它不为空，则执行 301 永久重定向。  #---------》 重点在这
    # 这会捕获通过 HTTPS 或已经转换为 HTTPS 的请求。				#
    if ($redirect_students_url != "") {
        return 301 $redirect_students_url;				#
    }									#
    # ================================================			#

    ssl_certificate         /www/server/panel/vhost/cert/wallpaper.kivotos.de/fullchain.pem;
    ssl_certificate_key     /www/server/panel/vhost/cert/wallpaper.kivotos.de/privkey.pem;
    ssl_protocols TLSv1.1 TLSv1.2 TLSv1.3;
    ssl_ciphers EECDH+CHACHA20:EECDH+CHACHA20-draft:EECDH+AES128:RSA+AES128:EECDH+AES256:RSA+AES256:EECDH+3DES:RSA+3DES:!MD5;
    ssl_prefer_server_ciphers on;
    ssl_session_tickets on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;
    add_header Strict-Transport-Security "max-age=31536000";
    add_header Alt-Svc 'quic=":443"; h3=":443"; h3-29=":443"; h3-27=":443";h3-25=":443"; h3-T050=":443"; h3-Q050=":443";h3-Q049=":443";h3-Q048=":443"; h3-Q046=":443"; h3-Q043=":443"';
    error_page 497  https://$host$request_uri;

    #SSL-END

    #ERROR-PAGE-START  错误页配置，可以注释、删除或修改
    error_page 404 /404.html;
    #error_page 502 /502.html;
    #ERROR-PAGE-END

    #PHP-INFO-START  PHP引用配置，可以注释或修改
    include enable-php-80.conf;
    #PHP-INFO-END

    #REWRITE-START URL重写规则引用,修改后将导致面板设置的伪静态规则失效
    include /www/server/panel/vhost/rewrite/wallpaper.kivotos.de.conf;
    #REWRITE-END

    #禁止访问的文件或目录
    location ~ ^/(\.user.ini|\.htaccess|\.git|\.env|\.svn|\.project|LICENSE|README.md)
    {
        return 404;
    }

    #一键申请SSL证书验证目录相关设置
    location ~ \.well-known{
        allow all;
    }

    #禁止在证书验证目录放入敏感文件
    if ( $uri ~ "^/\.well-known/.*\.(php|jsp|py|js|css|lua|ts|go|zip|tar\.gz|rar|7z|sql|bak)$" ) {
        return 403;
    }

    location ~ .*\.(gif|jpg|jpeg|png|bmp|swf)$
    {
        expires         30d;
        error_log /dev/null;
        access_log /dev/null;
    }

    location ~ .*\.(js|css)?$
    {
        expires         12h;
        error_log /dev/null;
        access_log /dev/null;
    }
    access_log  /www/wwwlogs/wallpaper.kivotos.de.log;
    error_log   /www/wwwlogs/wallpaper.kivotos.de.error.log;
}
